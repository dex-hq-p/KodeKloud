## 2. Upgrade control plane len 1.37.x
```
vi /etc/apt/sources.list.d/kubernetes.list. [ edit 1.36 -> 1.37]
sudo apt update

sudo apt-cache madison kubeadm

sudo apt-mark unhold kubeadm && \
sudo apt-get update && sudo apt-get install -y kubeadm='1.37.x-*' && \
sudo apt-mark hold kubeadm

kubeadm version

sudo kubeadm upgrade plan

sudo kubeadm upgrade apply v1.37.x

sudo apt-mark unhold kubelet kubectl && \
sudo apt-get update && sudo apt-get install -y kubelet='1.37.x-*' kubectl='1.37.x-*' && \
sudo apt-mark hold kubelet kubectl

sudo systemctl daemon-reload
sudo systemctl restart kubelet
```

####  Upgrade kubelet trên node `worker-01`.
```
kubectl drain worker-01 --ignore-daemonsets

vi /etc/apt/sources.list.d/kubernetes.list. -> sửa 1.36 -> 1.37
sudo apt update

sudo apt-cache madison 

sudo apt-mark unhold kubeadm kubectl kubelet && \
sudo apt-get update && sudo apt-get install -y kubeadm='1.37.x-*' kubectl='1.37.x-*' && kubelet='1.37.x-*' \
sudo apt-mark hold kubeadm kubectl

sudo kubeadm upgrade node

sudo systemctl daemon-reload
sudo systemctl restart kubelet
```
##### Đảm bảo các workload được khôi phục.
```
kubectl uncordon worker-01
```
